```python
import aiosqlite
import os

DB_PATH = os.path.join(os.path.dirname(__file__), "chat.db")

async def get_db():
    return await aiosqlite.connect(DB_PATH)

async def init_db():
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute("""
            CREATE TABLE IF NOT EXISTS groups (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        await db.execute("""
            CREATE TABLE IF NOT EXISTS members (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                group_id INTEGER NOT NULL,
                name TEXT NOT NULL,
                type TEXT NOT NULL CHECK(type IN ('human', 'bot')),
                role TEXT,
                system_prompt TEXT,
                avatar_color TEXT DEFAULT '#6366f1',
                FOREIGN KEY (group_id) REFERENCES groups(id)
            )
        """)
        await db.execute("""
            CREATE TABLE IF NOT EXISTS messages (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                group_id INTEGER NOT NULL,
                member_id INTEGER NOT NULL,
                content TEXT NOT NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (group_id) REFERENCES groups(id),
                FOREIGN KEY (member_id) REFERENCES members(id)
            )
        """)
        await db.commit()

async def get_group(db, group_id: int):
    async with db.execute("SELECT * FROM groups WHERE id = ?", (group_id,)) as cur:
        row = await cur.fetchone()
        if row:
            return {"id": row[0], "name": row[1], "created_at": row[2]}
        return None

async def get_members(db, group_id: int):
    async with db.execute("SELECT * FROM members WHERE group_id = ?", (group_id,)) as cur:
        rows = await cur.fetchall()
        return [{"id": r[0], "group_id": r[1], "name": r[2], "type": r[3],
                 "role": r[4], "system_prompt": r[5], "avatar_color": r[6]} for r in rows]

async def get_messages(db, group_id: int, limit: int = 50):
    async with db.execute("""
        SELECT m.id, m.group_id, m.member_id, m.content, m.created_at,
               mb.name, mb.type, mb.avatar_color
        FROM messages m
        JOIN members mb ON m.member_id = mb.id
        WHERE m.group_id = ?
        ORDER BY m.created_at DESC LIMIT ?
    """, (group_id, limit)) as cur:
        rows = await cur.fetchall()
        return list(reversed([{
            "id": r[0], "group_id": r[1], "member_id": r[2], "content": r[3],
            "created_at": r[4], "sender_name": r[5], "sender_type": r[6],
            "avatar_color": r[7]
        } for r in rows]))

async def save_message(db, group_id: int, member_id: int, content: str):
    async with db.execute(
        "INSERT INTO messages (group_id, member_id, content) VALUES (?, ?, ?)",
        (group_id, member_id, content)
    ) as cur:
        await db.commit()
        return cur.lastrowid

async def get_member(db, member_id: int):
    async with db.execute("SELECT * FROM members WHERE id = ?", (member_id,)) as cur:
        row = await cur.fetchone()
        if row:
            return {"id": row[0], "group_id": row[1], "name": row[2], "type": row[3],
                    "role": row[4], "system_prompt": row[5], "avatar_color": row[6]}
        return None
```

from fastapi import WebSocket
from typing import Dict, List

class WSManager:
    def __init__(self):
        # group_id -> list of (websocket, member_id)
        self.connections: Dict[int, List[tuple]] = {}

    async def connect(self, websocket: WebSocket, group_id: int, member_id: int):
        await websocket.accept()
        if group_id not in self.connections:
            self.connections[group_id] = []
        self.connections[group_id].append((websocket, member_id))

    def disconnect(self, websocket: WebSocket, group_id: int):
        if group_id in self.connections:
            self.connections[group_id] = [
                (ws, mid) for ws, mid in self.connections[group_id] if ws != websocket
            ]

    async def broadcast(self, group_id: int, message: dict):
        if group_id not in self.connections:
            return
        dead = []
        for ws, _ in self.connections[group_id]:
            try:
                await ws.send_json(message)
            except Exception:
                dead.append(ws)
        for ws in dead:
            self.disconnect(ws, group_id)

manager = WSManager()
```

import os
import httpx
from dotenv import load_dotenv

load_dotenv()

DEEPSEEK_API_KEY = os.getenv("DEEPSEEK_API_KEY")
DEEPSEEK_URL = "https://api.deepseek.com/v1/chat/completions"

async def call_ai(system_prompt: str, history: list[dict], user_message: str) -> str:
    """
    抽象 AI 调用接口，未来可替换为其他模型。
    history: [{"role": "user/assistant", "content": "..."}]
    """
    messages = [{"role": "system", "content": system_prompt}]
    messages.extend(history[-10:])  # 保留最近 10 条上下文
    messages.append({"role": "user", "content": user_message})

    async with httpx.AsyncClient(timeout=60) as client:
        response = await client.post(
            DEEPSEEK_URL,
            headers={"Authorization": f"Bearer {DEEPSEEK_API_KEY}"},
            json={
                "model": "deepseek-chat",
                "messages": messages,
                "temperature": 0.7,
                "max_tokens": 1000,
            }
        )
        response.raise_for_status()
        return response.json()["choices"][0]["message"]["content"]
```

```python
from typing import Optional

def should_bot_respond(message: str, bot_name: str, bot_role: str) -> bool:
    """
    判断某个 bot 是否应该响应这条消息。
    规则：@mention bot名字，或者消息包含与该角色相关的关键词。
    """
    msg_lower = message.lower()
    name_lower = bot_name.lower()

    # @mention 触发
    if f"@{name_lower}" in msg_lower:
        return True

    # 角色关键词触发
    role_keywords = {
        "需求分析师": ["需求", "分析", "功能", "要求", "规格"],
        "开发工程师": ["开发", "代码", "实现", "编程", "bug", "功能"],
        "测试工程师": ["测试", "bug", "质量", "用例", "验证"],
    }

    keywords = role_keywords.get(bot_role, [])
    return any(kw in msg_lower for kw in keywords)

def build_context_message(
    message: str,
    sender_name: str,
    recent_messages: list[dict]
) -> tuple[list[dict], str]:
    """
    构建发给 AI 的历史上下文和当前消息。
    """
    history = []
    for msg in recent_messages[-8:]:
        role = "assistant" if msg["sender_type"] == "bot" else "user"
        history.append({
            "role": role,
            "content": f"[{msg['sender_name']}]: {msg['content']}"
        })

    user_message = f"[{sender_name}]: {message}"
    return history, user_message
```

from fastapi import FastAPI, WebSocket, WebSocketDisconnect, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from contextlib import asynccontextmanager
import json

from database import init_db, get_db, get_group, get_members, get_messages, save_message, get_member
from ws_manager import manager
from ai_client import call_ai
from role_router import should_bot_respond, build_context_message
from models import SendMessageRequest, AddMemberRequest, CreateGroupRequest

@asynccontextmanager
async def lifespan(app: FastAPI):
    await init_db()
    yield

app = FastAPI(lifespan=lifespan)

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

# ── REST API ──────────────────────────────────────────────

@app.get("/api/groups/{group_id}")
async def get_group_info(group_id: int):
    async with await get_db() as db:
        group = await get_group(db, group_id)
        if not group:
            raise HTTPException(404, "Group not found")
        members = await get_members(db, group_id)
        return {"group": group, "members": members}

@app.get("/api/groups/{group_id}/messages")
async def get_group_messages(group_id: int):
    async with await get_db() as db:
        return await get_messages(db, group_id)

@app.post("/api/groups")
async def create_group(req: CreateGroupRequest):
    async with await get_db() as db:
        async with db.execute("INSERT INTO groups (name) VALUES (?)", (req.name,)) as cur:
            await db.commit()
            group_id = cur.lastrowid
        return {"id": group_id, "name": req.name}

@app.post("/api/groups/{group_id}/members")
async def add_member(group_id: int, req: AddMemberRequest):
    async with await get_db() as db:
        async with db.execute(
            "INSERT INTO members (group_id, name, type, role, system_prompt, avatar_color) VALUES (?,?,?,?,?,?)",
            (group_id, req.name, req.type, req.role, req.system_prompt, req.avatar_color)
        ) as cur:
            await db.commit()
            return {"id": cur.lastrowid, "name": req.name, "type": req.type}

# ── WebSocket ──────────────────────────────────────────────

@app.websocket("/ws/{group_id}/{member_id}")
async def websocket_endpoint(websocket: WebSocket, group_id: int, member_id: int):
    await manager.connect(websocket, group_id, member_id)
    try:
        while True:
            data = await websocket.receive_text()
            payload = json.loads(data)
            content = payload.get("content", "").strip()
            if not content:
                continue

            async with await get_db() as db:
                sender = await get_member(db, member_id)
                if not sender:
                    continue

                # 保存并广播人类消息
                msg_id = await save_message(db, group_id, member_id, content)
                recent = await get_messages(db, group_id)

                await manager.broadcast(group_id, {
                    "type": "message",
                    "id": msg_id,
                    "member_id": member_id,
                    "sender_name": sender["name"],
                    "sender_type": sender["type"],
                    "avatar_color": sender["avatar_color"],
                    "content": content,
                })

                # 触发 AI 角色响应
                bots = [m for m in await get_members(db, group_id) if m["type"] == "bot"]
                for bot in bots:
                    if not should_bot_respond(content, bot["name"], bot["role"] or ""):
                        continue

                    history, user_msg = build_context_message(content, sender["name"], recent)
                    system_prompt = bot["system_prompt"] or f"你是{bot['name']}，{bot['role']}。"

                    # 广播"正在输入"状态
                    await manager.broadcast(group_id, {
                        "type": "typing",
                        "sender_name": bot["name"],
                        "avatar_color": bot["avatar_color"],
                    })

                    ai_reply = await call_ai(system_prompt, history, user_msg)

                    bot_msg_id = await save_message(db, group_id, bot["id"], ai_reply)
                    await manager.broadcast(group_id, {
                        "type": "message",
                        "id": bot_msg_id,
                        "member_id": bot["id"],
                        "sender_name": bot["name"],
                        "sender_type": "bot",
                        "avatar_color": bot["avatar_color"],
                        "content": ai_reply,
                    })

    except WebSocketDisconnect:
        manager.disconnect(websocket, group_id)
```

import { useEffect, useRef, useCallback } from 'react'

export function useWebSocket(groupId, memberId, onMessage) {
  const ws = useRef(null)

  useEffect(() => {
    if (!groupId || !memberId) return
    const url = `ws://localhost:8000/ws/${groupId}/${memberId}`
    ws.current = new WebSocket(url)
    ws.current.onmessage = (e) => onMessage(JSON.parse(e.data))
    ws.current.onerror = (e) => console.error('WS error', e)
    return () => ws.current?.close()
  }, [groupId, memberId])

  const send = useCallback((content) => {
    if (ws.current?.readyState === WebSocket.OPEN) {
      ws.current.send(JSON.stringify({ content }))
    }
  }, [])

  return { send }
}
```

import { useEffect, useState, useRef } from 'react'
import { fetchGroupInfo, fetchMessages } from '../api'
import { useWebSocket } from '../hooks/useWebSocket'
import MemberList from './MemberList'
import MessageBubble from './MessageBubble'
import MessageInput from './MessageInput'

const GROUP_ID = 1

export default function ChatWindow({ memberId, memberName }) {
  const [group, setGroup] = useState(null)
  const [members, setMembers] = useState([])
  const [messages, setMessages] = useState([])
  const [typing, setTyping] = useState(null) // {sender_name, avatar_color}
  const bottomRef = useRef(null)

  useEffect(() => {
    fetchGroupInfo(GROUP_ID).then(({ group, members }) => {
      setGroup(group)
      setMembers(members)
    })
    fetchMessages(GROUP_ID).then(setMessages)
  }, [])

  const handleWsMessage = (data) => {
    if (data.type === 'typing') {
      setTyping({ sender_name: data.sender_name, avatar_color: data.avatar_color })
    } else if (data.type === 'message') {
      setTyping(null)
      setMessages((prev) => [...prev, data])
    }
  }

  const { send } = useWebSocket(GROUP_ID, memberId, handleWsMessage)

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth' })
  }, [messages, typing])

  return (
    <div className="flex h-screen bg-gray-900">
      <MemberList members={members} />
      <div className="flex flex-col flex-1 min-w-0">
        {/* 头部 */}
        <div className="h-14 bg-gray-900 border-b border-gray-700 flex items-center px-4 gap-2">
          <span className="text-gray-300 font-semibold"># {group?.name || '加载中...'}</span>
          <span className="text-xs text-gray-500 ml-2">· {members.length} 名成员</span>
        </div>
        {/* 消息列表 */}
        <div className="flex-1 overflow-y-auto px-4 py-4 space-y-1">
          {messages.map((msg) => (
            <MessageBubble key={msg.id} msg={msg} />
          ))}
          {typing && <MessageBubble msg={typing} isTyping />}
          <div ref={bottomRef} />
        </div>
        <MessageInput onSend={send} />
      </div>
    </div>
  )
}
